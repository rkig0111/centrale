INSTALLATION DE L'ENVIRONNEMENT DE TRAVAIL:
EN VRAC!!!

adaptez tous les PATH en conséquence si besoin!

tous mes projets sont dans ~/developpement/:
dans votre répertoire home
    mkdir developpement
    cd developpement
    git clone https://github.com/rkig0111/centrale.git
aller dans le répertoire nouvellemnt créé :
    cd centrale
y créer l'environnement virtuel :
    python3 -m venv .centrale
l'activer :
    source .centrale/bin/activate
créer toutes les dépendances que nous aurons besoin :
    pip install -r requirements.txt 

    sudo apt-get install chromium-chromedriver
# pour savoir où se trouve l'executable chromedriver, on en a besoin dans centrale.py (variable 'service')
    which chromedriver   

# il faut adapter le fichier 'centrale_a_modifier.ini' à sa configuration et le renommer 'centrale.ini'. Il centralise un peu toutes les variables dont ont a besoin.
# c'est assez explicite. besoin surtout de l' adresse IP de la centrale et des codes d'accès pour s'y connecter.

Dans le script 'centrale.py' j'ai essayé de documenter un max de choses en allant.
j'espère seulement être le plus compréhensible possible..  ;-)

pip install python-memcached

                                               'Marche A', 'Marche B', 'Marche C', 'Marche A B C', 'Arrêt A B C,' 'Test'

import memcache
shared = memcache.Client(['127.0.0.1:11211'], debug=0) 
cde = shared.get('OnOff')   
print(cde)

######   INSTALL NGINX    #######

sudo apt update
sudo apt upgrade
sudo apt install nginx
sudo systemctl enable --now nginx         # démarrer et activer au boot

sudo mkdir -p /var/www/centrale
echo '<h1>Hello Nginx!</h1>' | sudo tee /var/www/centrale/index.html

sudo tee /etc/nginx/conf.d/centrale.conf << 'EOF'
server {
    listen 80;
    server_name centrale.local;

    root /var/www/centrale;
    index index.html;

    location / {
        try_files $uri $uri/ =404;
    }
}
EOF

pour accéder à votre Raspberry pi de l'extérieur avec votre nom de domaine (entre crochet)

sudo tee /etc/nginx/sites-available/[xxxxxx.freeboxos.fr] << 'EOF'      
server {
    listen 80;
    server_name centrale.local;

    root /var/www/centrale;
    index index.html;

    location / {
        try_files $uri $uri/ =404;
    }
}
EOF

import os
import RPi.GPIO as GPIO
from flask import Flask, render_template, Response
import datetime

GPIO.setmode(GPIO.BCM)
dataPin=[i for i in range(2,28)]
for dp in dataPin: GPIO.setup(dp,GPIO.IN)#,pull_up_down=GPIO.PUD_UP)

data=[]
now=datetime.datetime.now()
timeString=now.strftime("%Y-%m-%d %H:%M")
templateData={
    'title':'Raspberry Pi 3B+ Web Controller',
    'time':timeString,
    'data':data,
}

def getData():
    data=[]
    for i,dp in enumerate(dataPin): data.append(GPIO.input(dataPin[i]))
    
    return data

app=Flask(__name__)
    
@app.route('/')
def index():
    #return 'hello world!'
    now=datetime.datetime.now()
    timeString=now.strftime("%Y-%m-%d %H:%M")
    data=getData()
    templateData={
        'title':'Raspberry Pi 3B+ Web Controller',
        'time':timeString,
        'data':data,
    }
    #return render_template('rpi_index.html',**templateData)
    return render_template('rpi3b_webcontroller.html',**templateData)           

@app.route('/<actionid>') 
def handleRequest(actionid):
    print("Button pressed : {}".format(actionid))
    return "OK 200"
